2. Display Product Name and CPU
-------------------------------

db.products.find(
    {},
    {
        _id: 0,
        name: 1,
        "spec.cpu": 1
    }
)


3. Display Second Storage Value Using Aggregation
-------------------------------------------------

db.products.aggregate([
{
    $project: {
        _id: 0,
        name: 1,
        secondStorage: {
            $arrayElemAt: ["$storage", 1]
        }
    }
}
])


4. Find Products with Screen Size Less Than 7
---------------------------------------------

db.products.find(
{
    "spec.screen": {
        $lt: 7
    }
},
{
    _id: 0,
    name: 1,
    "spec.screen": 1
}
)


5. Find Products Having Storage Less Than 128
---------------------------------------------

db.products.find(
{
    storage: {
        $lt: 128
    }
},
{
    _id: 0,
    name: 1,
    storage: 1
}
)
